```python title="DAG and tasks" wrap
from airflow.sdk import dag, task
from pendulum import datetime

@task
def scrape_data(query: str) -> dict: ...

@dag("weather_scraper", schedule="*/20 * * * *",
     start_date=datetime(2026, 1, 1),
     catchup=False, tags=["weather"])
def main(query: str = "kosice,sk"):
    data = scrape_data(query)
    is_service_alive() >> data

if __name__ == "__main__":
    main().test()
else:
    main()
```

```python title="Connections, Variables, errors" wrap
from airflow.sdk import BaseHook, Variable
from airflow.sdk.exceptions import (
    AirflowException, AirflowFailException)

conn = BaseHook.get_connection("openweathermap")
conn.host; conn.port; conn.password
conn.extra_dejson.get("units")
Variable.get("weather_city", default="kosice,sk")
Variable.get("cfg", deserialize_json=True)

@task(retries=3, retry_delay=timedelta(seconds=30))
raise AirflowException("retry me")
raise AirflowFailException("no retry")
```

</div>

| What | How |
| --- | --- |
| cron | `min hour day month weekday` · `*/20 * * * *` · `@hourly`, `@daily`, `@weekly`, `@monthly`, `@yearly` |
| config via environment | `AIRFLOW__CORE__LOAD_EXAMPLES=False` |
| Variable / Connection from environment | `AIRFLOW_VAR_WEATHER_CITY=…` · `AIRFLOW_CONN_OPENWEATHERMAP='{"conn_type": …}'` |
| task states | queued (grey) · running · success (green) · failed (red) · upstream_failed (orange) · up_for_retry (yellow) |
| `retries=3` | up to 4 runs; `retry_delay` defaults to 300 s |
| path to the data | `Path(__file__).parent.parent / "data"` in `dags/constants.py` |
| JSON schema | `jsonschema.validate(instance=data, schema=schema)` |
| logging | `logger = logging.getLogger(__name__)` · DEBUG 10 < INFO 20 < WARNING 30 < ERROR 40 < CRITICAL 50 · `print` = INFO |
| swap | `fallocate -l 2G /swapfile` → `chmod 600` → `mkswap` → `swapon` · `free -h` |
| tmux (prefix `Ctrl+b`) | `c` new window · `n`/`p` next/previous · `"`/`%` split · `d` detach · `tmux attach` |

```python title="Parallel healthchecks + retries" wrap
from datetime import timedelta
from sh import ping                 # pip install sh

@task.bash
def ping_owm() -> str:
    return "ping -c 1 -w 2 api.openweathermap.org"

@task(task_display_name="Ping (sh)", retries=3,
      retry_delay=timedelta(seconds=10))
def ping_sh() -> None:
    ping("-c", 1, "-w", 2, "api.openweathermap.org")

@task(task_display_name="RustFS Healthcheck")
def is_rustfs_alive() -> None:
    conn = BaseHook.get_connection("rustfs")
    url = f"{conn.schema}://{conn.host}:{conn.port}"
    r = httpx.get(f"{url}/health")
    if r.status_code != HTTPStatus.OK:
        raise AirflowFailException("RustFS unhealthy")

# in main(): a list = parallel
data = [ping_owm(), ping_sh(), is_rustfs_alive()] \
    >> scrape_data(query)
```

## Task groups, S3, and assets

<div class="sheet-grid">

```python title="Task groups" wrap
from airflow.sdk import task_group

@task_group(group_id="tg_healthcheck",
            group_display_name="Healthcheck",
            tooltip="Healthcheck of external services.")
def tg_healthcheck():
    return [ping_owm(), is_rustfs_alive()]

@task_group(group_display_name="Weather Ingestion")
def tg_weather_ingestion(query: str):
    publish_data(process_data(scrape_data(query)))
    # return nothing if you chain the group with >>

tg_healthcheck() >> tg_weather_ingestion(query)
```

```python title="boto3: bucket" wrap
import boto3

storage = boto3.resource(
    "s3", endpoint_url="http://<host>:9000",
    aws_access_key_id="<ACCESS_KEY>",
    aws_secret_access_key="<SECRET_KEY>")
bucket = storage.Bucket("mirek")  # no network
```

```python title="Assets" wrap

    uri="s3://datasets/dataset.csv",
```
